This file is the single source of the Emacs configuration.  Nix tangles it at
build time (see =default.nix=) into =~/.emacs.d/early-init.el= and
=~/.emacs.d/init.el=.  Do not edit the generated files directly.

Values that contain Nix store paths (=my/nix-tool-path=, =my/nix-coreutils-ls=,
=my/nix-wakatime-cli=, =my/nix-rust-src-path=) are generated by =default.nix= into =nix-vars.el=.
Packages are installed by Nix via =programs.emacs.extraPackages=; when you use
a new package here, add it there too.

#+begin_src emacs-lisp
(dolist (char/ligature-re
         `((?-  . ,(rx (or (or "-->" "-<<" "->>" "-|" "-~" "-<" "->") (+ "-"))))
           (?/  . ,(rx (or (or "/==" "/=" "/>" "/**" "/*") (+ "/"))))
           (?*  . ,(rx (or (or "*>" "*/") (+ "*"))))
           (?<  . ,(rx (or (or "<<=" "<<-" "<|||" "<==>" "<!--" "<=>" "<||" "<|>" "<-<"
                               "<==" "<=<" "<-|" "<~>" "<=|" "<~~" "<$>" "<+>" "</>"
                               "<*>" "<->" "<=" "<|" "<:" "<>" "<$" "<-" "<~" "<+"
                               "</" "<*")
                           (+ "<"))))
           (?:  . ,(rx (or (or ":?>" "::=" ":>" ":<" ":?" ":=") (+ ":"))))
           (?=  . ,(rx (or (or "=>>" "==>" "=/=" "=!=" "=>" "=:=") (+ "="))))
           (?!  . ,(rx (or (or "!==" "!=") (+ "!"))))
           (?>  . ,(rx (or (or ">>-" ">>=" ">=>" ">]" ">:" ">-" ">=") (+ ">"))))
           (?&  . ,(rx (+ "&")))
           (?|  . ,(rx (or (or "|->" "|||>" "||>" "|=>" "||-" "||=" "|-" "|>"
                               "|]" "|}" "|=")
                           (+ "|"))))
           (?.  . ,(rx (or (or ".?" ".=" ".-" "..<") (+ "."))))
           (?+  . ,(rx (or "+>" (+ "+"))))
           (?\[ . ,(rx (or "[<" "[|")))
           (?\{ . ,(rx "{|"))
           (?\? . ,(rx (or (or "?." "?=" "?:") (+ "?"))))
           (?#  . ,(rx (or (or "#_(" "#[" "#{" "#=" "#!" "#:" "#_" "#?" "#(")
                           (+ "#"))))
           (?\; . ,(rx (+ ";")))
           (?_  . ,(rx (or "_|_" "__")))
           (?~  . ,(rx (or "~~>" "~~" "~>" "~-" "~@")))
           (?$  . ,(rx "$>"))
           (?^  . ,(rx "^="))
           (?\] . ,(rx "]#"))))
  (let ((char (car char/ligature-re))
        (ligature-re (cdr char/ligature-re)))
    (set-char-table-range composition-function-table char
                          `([,ligature-re 0 font-shape-gstring]))))

Web search helpers.

#+begin_src emacs-lisp
(defun my/web-search (base &optional prompt)
  "Search BASE (a URL ending so the query can be appended) with the
active region, or a prompted string (PROMPT) when there's no region."
  (browse-url
   (concat base
           (url-hexify-string
            (if (use-region-p)
                (buffer-substring-no-properties (region-beginning) (region-end))
              (read-string (or prompt "Search: ")))))))

(defun my/google ()
  "Search Google."
  (interactive)
  (my/web-search "https://www.google.com/search?q=" "Google: "))

(defun my/marginalia ()
  "Search Marginalia (indie-web search engine)."
  (interactive)
  (my/web-search "https://marginalia-search.com/search?query=" "Marginalia: "))

(defun my/browse-url-at-point-or-prompt ()
  "Open the URL at point in xwidget, or prompt if there isn't one."
  (interactive)
  (let ((url (browse-url-url-at-point)))
    (if url
        (browse-url url)
      (browse-url (read-string "URL: ")))))
#+end_src

** Rust

Built-in =rust-ts-mode= with rust-analyzer.  The toolchain (=rustc=, =cargo=,
=rustfmt=, =clippy=, =rust-analyzer=) comes from Nix; a project with its own
toolchain in =flake.nix= / =.envrc= overrides it through envrc.
rust-analyzer needs the standard library sources, which Nix provides via
=my/nix-rust-src-path=.  Formatting on save goes through rust-analyzer →
rustfmt, and diagnostics use clippy (see =eglot-workspace-configuration=).

#+begin_src emacs-lisp

* Claude Code

The Claude Code CLI in a Vterm buffer, one session per project, connected to
Emacs over MCP.  =C-c C= opens the transient menu, =claude-code-ide= starts a
session and =claude-code-ide-send-prompt= sends a prompt.  (=C-c c= is
=org-capture=.)

=claude-code-ide-cli-path= points at the Nix-wrapped binary, so a session
started here gets the MCP servers and agents from =modules/aspects/llm/claude=.
Emacs launched from the Dock does not read the login shell, so the plain
=claude= is not on its =PATH=.

The same goes for =NODE_EXTRA_CA_CERTS=: without it the CLI fails TLS
verification, so it is set here from the value Home Manager exports to shells.

#+begin_src emacs-lisp
  (when (boundp 'my/nix-claude)
    (setq claude-code-ide-cli-path my/nix-claude))
  (when (and (boundp 'my/nix-ca-certs)
             (not (getenv "NODE_EXTRA_CA_CERTS")))
    (setenv "NODE_EXTRA_CA_CERTS" my/nix-ca-certs))
  (require 'claude-code-ide)
  (with-eval-after-load 'claude-code-ide
    (claude-code-ide-emacs-tools-setup)
    (global-set-key (kbd "C-c C") #'claude-code-ide-menu))
#+end_src
